# Allows you to create your own dynamic keys
# to be called in localization.
# defined_text -> this is it, we're defining the text
# text -> a discrete entry that can be picked to display in loc.
# trigger -> determines if a text entry will be picked or not.
# (The triggers need to be valid for the scope the key is called in
# (eg Root or From.From).)
# localization_key -> points to the localization key
# that'll be used if trigger passes

defined_text = { # Country
	name = worldmajors_category_desc
	text = {
		trigger = {
			has_country_flag = pisyapopa_flag
		}
		localization_key = worldmajors_category_desc
	}
	text = {
		trigger = {
			has_country_flag = agressor_flag
		}
		localization_key = worldmajors_category_desc2
	}
	text = {
		trigger = {
			NOT = {
				has_country_flag = agressor_flag
				has_country_flag = pisyapopa_flag
			}
		}
		localization_key = worldmajors_category_desc3
	}
	text = {
		localization_key = worldmajors_category_desc4
	}
}
defined_text = {
  name = "GetMyFlagCountriesList"
  text = {
   trigger = {
    any_country = {
     has_country_flag = my_special_flag
    }
   }
   localization_key = "MyFlag_List_Start"
  }
  text = {
   localization_key = "MyFlag_No_Countries"
  }
}

defined_text = {
  name = "MyFlag_Country_List_Builder"
  text = {
   trigger = {
    THIS = {
     has_country_flag = my_special_flag
    }
   }
   localization_key = "MyFlag_Country_Entry_With_Comma"
  }
  text = {
   localization_key = "MyFlag_Country_Entry_No_Comma"
  }
}
