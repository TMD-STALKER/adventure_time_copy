	desc = stw.1.d

		}

		}
	}
	option = { 
		name = stw.1.b
		ai_chance = {
			base = 1
		}
		if = {
			limit = {
				country_exists = SOK
				SOK = { is_ai = yes }
			}
			
			SOK = {
				country_event = stw.3
				declare_war_on = {
					target = ROOT
					type = annex_everything
				}
			}
		}
		else_if = {
			limit = {
				country_exists = SOK
				SOK = { is_ai = no }
			}
			
			SOK = {
				country_event = stw.3
				create_wargoal = {
					target = ROOT
					type = annex_everything
				}
			}
		}
		else = {
			random_other_country = {
				limit = {
					any_owned_state = { is_core_of = SOK }
				}
				country_event = stw.3
				release_on_controlled = SOK
			}
			hidden_effect = {
				SOK = {
					declare_war_on = {
						target = ROOT
						type = annex_everything
					}
					division_template = {
						name = "monsters"

						division_names_group = PBB_MYD_01
						is_locked = no
						regiments = {
							infantry = { x = 0 y = 0 }
							infantry = { x = 0 y = 1 }
							infantry = { x = 0 y = 2 }
							infantry = { x = 0 y = 3 }
						}
					}
					capital_scope = {
						create_unit = {
							division = "name = \"monsters\" division_template = \"monsters\" start_manpower_factor = 1"
							owner = SOK
							count = 5
							allow_spawning_on_enemy_provs = yes
						}
					}
				}
			}
		}
		
		add_timed_idea = {
			idea = STW_war_scouting
			days = 180
		}
	}
	option = { 
		name = stw.1.c
		ai_chance = {
			base = 1
		}
		random_other_country = {
			limit = {
				any_owned_state = { is_core_of = SOK }
			}
			country_event = stw.3
		}
		add_timed_idea = {
			idea = STW_war_scouting2
			days = 180
		}
		start_border_war = {
			change_state_after_war = yes
			attacker = {
				state = 195
				num_provinces = 1
				on_win = at_border.1.t
				on_lose = at_border.2.t
				on_cancel = at_border.2.t
				modifier = 0.1
				dig_in_factor = 0
				terrain_factor = 0
			}	
			defender = {
				state = 178
				num_provinces = 1
				on_win = at_border.3.t
				on_lose = at_border.4.t
				on_cancel = at_border.4.t
			}
		}
	}

}
